cmake_minimum_required(VERSION 3.22)
project(Lexicon224 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LEXICON_DSP_ONLY "Build portable DSP tests without JUCE" OFF)
include(CTest)
add_library(reverb224 Source/DSP/Reverb224.cpp Source/Programs/ProgramBank.cpp)
target_include_directories(reverb224 PUBLIC Source)
set_target_properties(reverb224 PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(BUILD_TESTING)
 add_executable(dsp_tests Tests/DspTests.cpp)
 target_link_libraries(dsp_tests PRIVATE reverb224)
 add_test(NAME dsp_tests COMMAND dsp_tests)
 add_executable(render_responses Tests/RenderResponses.cpp)
 target_link_libraries(render_responses PRIVATE reverb224)
endif()
if(LEXICON_DSP_ONLY)
 return()
endif()
set(JUCE_PATH "" CACHE PATH "Local JUCE 7 checkout")
if(JUCE_PATH)
 add_subdirectory("${JUCE_PATH}" juce)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(Lexicon224 COMPANY_NAME "AulaMix" BUNDLE_ID "com.aulamix.reverb224"
 PLUGIN_MANUFACTURER_CODE AuMx PLUGIN_CODE L224 FORMATS VST3 PRODUCT_NAME "Lexicon224"
 EDITOR_WANTS_KEYBOARD_FOCUS TRUE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(Lexicon224)
juce_add_binary_data(PresetData NAMESPACE PresetData HEADER_NAME PresetData.h SOURCES ../presets/catalog.json)
target_link_libraries(Lexicon224 PRIVATE PresetData)
target_sources(Lexicon224 PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/Programs/RegisterStore.cpp Source/Programs/PresetLibrary.cpp Source/UI/PresetBrowser.cpp)
target_compile_definitions(Lexicon224 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Lexicon224 PRIVATE reverb224 juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(BUILD_TESTING)
 add_executable(plugin_tests Tests/PluginTests.cpp)
 target_link_libraries(plugin_tests PRIVATE Lexicon224)
 target_include_directories(plugin_tests PRIVATE Source $<TARGET_PROPERTY:Lexicon224,INCLUDE_DIRECTORIES>)
 add_test(NAME plugin_tests COMMAND plugin_tests)
 add_executable(render_browser Tests/RenderBrowser.cpp)
 target_link_libraries(render_browser PRIVATE Lexicon224)
 target_include_directories(render_browser PRIVATE Source $<TARGET_PROPERTY:Lexicon224,INCLUDE_DIRECTORIES>)
endif()
